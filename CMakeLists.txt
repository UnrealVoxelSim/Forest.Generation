cmake_minimum_required(VERSION 3.30)
cmake_policy(SET CMP0168 NEW)
project(UnrealVoxelSim.Forest.Generation VERSION 0.1.0 LANGUAGES CXX)
include(FetchContent)
set(UNREALVOXELSIM_BUILD_CMAKE_SOURCE "" CACHE PATH "Optional local source override for Build.CMake.")
get_filename_component(unrealVoxelSimWorkspaceRoot "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(NOT UNREALVOXELSIM_BUILD_CMAKE_SOURCE AND EXISTS "${unrealVoxelSimWorkspaceRoot}/Build.CMake/UnrealVoxelSim.Build.CMake.cmake")
    set(UNREALVOXELSIM_BUILD_CMAKE_SOURCE "${unrealVoxelSimWorkspaceRoot}/Build.CMake" CACHE PATH "" FORCE)
endif()
if(NOT UNREALVOXELSIM_INTERNAL_SOURCE_ROOT AND EXISTS "${unrealVoxelSimWorkspaceRoot}/Build.CMake/CMakeLists.txt")
    set(UNREALVOXELSIM_INTERNAL_SOURCE_ROOT "${unrealVoxelSimWorkspaceRoot}" CACHE PATH "" FORCE)
endif()
FetchContent_Declare(uvsbuild SOURCE_DIR "${UNREALVOXELSIM_BUILD_CMAKE_SOURCE}")
FetchContent_MakeAvailable(uvsbuild)
include("${uvsbuild_SOURCE_DIR}/UnrealVoxelSim.Build.CMake.cmake")
UnrealVoxelSimAddInternalDependency(GITHUB_REPOSITORY UnrealVoxelSim/Forest.Generation.Api)
UnrealVoxelSimAddInternalDependency(GITHUB_REPOSITORY UnrealVoxelSim/Trees.Api)
UnrealVoxelSimAddInternalDependency(GITHUB_REPOSITORY UnrealVoxelSim/Voxel.Solid.Api)
UnrealVoxelSimConfigureModule()
